# Smithy AWS protocol test cases the AWS SDK fails today (see README.md).
# protocol-tests.test.ts fails on a failing case missing
# from this list, and on a case in this list that passes: delete its line
# when a fix makes it pass.
#
# One case per line: the case id and its kind, (request), (response) or
# (error), grouped by cause.

# `@default` values are not sent for members left out of a nested input
# structure. The AWS SDK for JavaScript skips these cases too ("JSv3 does not
# populate default values on the client side"), and sending them would add
# about 1,100 AWS input members to requests that leave them out today.
AwsJson10ClientPopulatesDefaultValuesInInput (request)
AwsJson10ClientPopulatesNestedDefaultValuesWhenMissing (request)
RestJsonClientPopulatesDefaultValuesInInput (request)
RestJsonClientPopulatesNestedDefaultValuesWhenMissing (request)

# An empty list or map element reads as an absent member; Smithy expects an
# empty list or map. packages/aws/test/protocols/xml.test.ts asserts the
# absent member for S3 and S3 Control.
QueryXmlEmptySelfClosedMaps (response)
XmlEmptyLists (response)
XmlEmptyMaps (response)
XmlEmptySelfClosedMaps (response)

# The expected body lists `DoubleDribble` before `Nested`, out of model
# order; the client writes members in model order, and this runner compares
# XML element order.
XmlNamespaceSimpleScalarProperties (request)

# An empty streaming body comes back as an empty stream, as in the AWS
# SDKs (S3 GetObject of an empty object); the case expects no member.
RestJsonStreamingTraitsWithNoBlobBody (response)
